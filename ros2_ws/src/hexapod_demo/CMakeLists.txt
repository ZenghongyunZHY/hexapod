cmake_minimum_required(VERSION 3.16)
project(hexapod_demo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
add_library(demo_core src/demo.cpp src/controller.cpp)
target_include_directories(demo_core PUBLIC include)
target_compile_options(demo_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
ament_target_dependencies(demo_core ament_index_cpp)
foreach(demo IN ITEMS stand_demo walk_demo)
  add_executable(${demo} src/${demo}.cpp)
  target_link_libraries(${demo} demo_core)
  install(TARGETS ${demo} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(demo_tests test/demo_tests.cpp)
  target_link_libraries(demo_tests demo_core)
  target_compile_options(demo_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME protocol_and_finite_sequence COMMAND demo_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/poses.csv)
endif()
ament_package()
